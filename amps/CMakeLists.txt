# Embeds every amp folder (amp.json + *.net) into a static library, so the plugin needs no files
# next to it. Adding an amp = adding a folder here and listing it below.
set(AES_AMP_FOLDERS jcm800_2203)

set(_gen "${CMAKE_CURRENT_BINARY_DIR}/embedded_amp_files.cpp")
set(_body "")
foreach(folder IN LISTS AES_AMP_FOLDERS)
  file(GLOB _files RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/${folder}" "${CMAKE_CURRENT_SOURCE_DIR}/${folder}/*")
  set(_entries "")
  foreach(f IN LISTS _files)
    set(_path "${CMAKE_CURRENT_SOURCE_DIR}/${folder}/${f}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_path}")
    file(READ "${_path}" _content)
    string(LENGTH "${_content}" _len)
    if(_len GREATER 16000)
      message(FATAL_ERROR "${folder}/${f} is ${_len} bytes: split it (MSVC string literals max out near 16 KB)")
    endif()
    string(APPEND _entries "      {\"${f}\", R\"AESFILE(${_content})AESFILE\"},\n")
  endforeach()
  string(APPEND _body "    {\"${folder}\", {\n${_entries}    }},\n")
endforeach()

file(WRITE "${_gen}" "// Generated by amps/CMakeLists.txt. Do not edit.
#include \"aes/embedded_amps.h\"

namespace aes {

const std::map<std::string, std::map<std::string, std::string>>& embeddedAmpFiles() {
  static const std::map<std::string, std::map<std::string, std::string>> files = {
${_body}  };
  return files;
}

}  // namespace aes
")

add_library(aes_amps STATIC "${_gen}" embedded_amps.cpp)
target_include_directories(aes_amps PUBLIC include)
target_link_libraries(aes_amps PUBLIC aes_engine)
